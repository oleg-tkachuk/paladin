{{ include "chart.fullname" . }} ({{ .Chart.AppVersion }}) installed in {{ .Release.Namespace }}.

PALADIN exposes three HTTP planes on one Pod (data / iam / admin). Pick how
to reach them:

{{- if .Values.httpRoute.enabled }}

  1. Gateway API (HTTPRoute)

     {{- if .Values.httpRoute.hostnames }}
     export APP_HOSTNAME={{ first .Values.httpRoute.hostnames }}
     {{- else }}
     export APP_HOSTNAME=$(kubectl -n {{(first .Values.httpRoute.parentRefs).namespace | default .Release.Namespace }} get gateway/{{ (first .Values.httpRoute.parentRefs).name }} -o jsonpath='{.spec.listeners[0].hostname}')
     {{- end }}
     {{- if and .Values.httpRoute.rules (first .Values.httpRoute.rules).matches (first (first .Values.httpRoute.rules).matches).path.value }}
     echo "Visit http://$APP_HOSTNAME{{ (first (first .Values.httpRoute.rules).matches).path.value }}"
     {{- end }}

{{- else if .Values.ingress.enabled }}

  1. Ingress

     {{- range $host := .Values.ingress.hosts }}
     {{- range .paths }}
     http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
     {{- end }}
     {{- end }}

{{- else if contains "NodePort" .Values.service.type }}

  1. NodePort

     export NODE_PORT=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "chart.fullname" . }} -o jsonpath='{.spec.ports[0].nodePort}')
     export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
     echo "data plane: http://$NODE_IP:$NODE_PORT"

{{- else if contains "LoadBalancer" .Values.service.type }}

  1. LoadBalancer

     NOTE: It may take a few minutes for the LB IP to be available.
     kubectl -n {{ .Release.Namespace }} get svc -w {{ include "chart.fullname" . }}

     export SERVICE_IP=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "chart.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
     echo "data plane: http://$SERVICE_IP:{{ .Values.service.ports.data }}"

{{- else }}

  1. Port-forward (one of the three planes at a time)

     export POD=$(kubectl -n {{ .Release.Namespace }} get pod -l app.kubernetes.io/name={{ include "chart.name" . }},app.kubernetes.io/instance={{ .Release.Name }} -o jsonpath='{.items[0].metadata.name}')

     # Data plane (object operations)
     kubectl -n {{ .Release.Namespace }} port-forward $POD {{ .Values.service.ports.data }}:{{ .Values.service.ports.data }}

     # IAM plane (login + token exchange)
     # kubectl -n {{ .Release.Namespace }} port-forward $POD {{ .Values.service.ports.iam }}:{{ .Values.service.ports.iam }}

     # Admin plane (internal/SRE)
     # kubectl -n {{ .Release.Namespace }} port-forward $POD {{ .Values.service.ports.admin }}:{{ .Values.service.ports.admin }}

{{- end }}

  2. Health check (any plane works — addProbes mounts on all three)

     curl http://<host>:{{ .Values.service.ports.data }}/livez

  3. Verify the role split (production-only — see docs/db-roles.md)

     kubectl -n {{ .Release.Namespace }} logs deploy/{{ include "chart.fullname" . }} | grep 'pool initialized'
